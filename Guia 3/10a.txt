MOV EAX, 0X01
MOV EDX, DS
LDH ECX, 4
LDL ECX, 1
MOV EFX, 0	; EFX = N

LEER:
SYS 0X1
CMP [EDX], -1
JZ FINLEER:
ADD EDX, 1
ADD EFX, 1
JMP LEER

FINLEER:
MOV AC, 0	; AC = i = 0
MOV EEX, EFX
SUB EEX, 1	; queremos recorrer N - 1 numeros
CICLO:
MOV EBX, DS
SYS 0X1		; leemos lo que ingresa el usuario para marcar como -1
CMP EEX, AC	; i = N - 1
JZ ESCRITURA:


SUBCICLO:
CMP [EBX], [EDX]
JZ ENCONTRADO
ADD EBX, 1
JMP SUBCICLO

ENCONTRADO:
MOV [EBX], -1
ADD AC, 1
JMP CICLO

ESCRITURA:
MOV EDX, DS
ESCCICLO:
CMP [EDX], -1
JNZ FIN
ADD EDX, 1
JMP ESCCICLO


FIN:
SYS 0X02
STOP
