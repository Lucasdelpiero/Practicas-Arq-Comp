MOV EAX, 0X01		; lee decimales
LDL ECX, 1		; lee de a 1 valor 1 bit
LDH ECX, 1
MOV EFX, 0;		; EFX numero final

LECTURA:
SYS 0x01		; lee del teclado
CMP EDX, 0		
JZ SHIFTEA		; Si es cero incrementa y lee de nuevo

CMP EDX, 1
JPZ SHIFTEA		; Si es uno incrementa y lee de nuevo
JNZ FIN			; Si no es cero finaliza

SHIFTEA:
SHL EFX, 1		; se shiftea todo lo acumulado hasta el momento			
ADD EFX, EDX		; se le suma el ultimo ingresado
JMP LECTURA

FIN:
MOV EDX, EFX
STOP
