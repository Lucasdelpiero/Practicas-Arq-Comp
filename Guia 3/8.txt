MOV EAX, 0X01
MOV EBX, 0	; cant de datos ingresados
MOV EDX, DS

LEER:
LDH ECX, 4
LDL ECX, 1
SYS 0X01

CMP [EDX], 0	; si ingresa negativo deja de agregar datos
JN FINLECTURA
ADD EDX, 4
JMP LEER;	; si no es negativo lo agrega

FINLECTURA:
MOV EBX, EDX	; EBX ahora guarda hasta donde hay que leer (skipea si llega aca)
MOV EDX, DS		; comienza a leer desde la posicion del DS

;---------CICLO EXTERNO

CICLONUMERO:
MOV ECX, 0X1		; mascara
CMP EDX, EBX		; si llego a una posicion mas alla del final de donde guarde numeros, termino
JZ FIN
MOV EFX, 0		; EFX = cant unos
MOV AC, 0		; AC = i
	
;---------CICLO INTERNO

CICLOBITS:
MOV EEX, [EDX]		; guardo el contenido
AND [EDX], ECX		; se hace mascara (luego se reasigna para no perderlo)

JZ NOSUMA		; si a la mascara da cero el (CC ZERO) se activa y no suma un uno
ADD EFX, 1		; si hay un 1 entonces suma 1

NOSUMA:
MOV [EDX], EEX		; reasigna contenido a EDX
ADD AC, 1
SHL ECX, 1		; shifteo mascara para recorrer todos los bits
CMP AC, 16		; recorro 16 bits
JNZ CICLOBITS

;---------FIN CICLO INTERNO

MOV EAX, 0X01		; me preparo para leer la cant de unos
LDH ECX, 4
LDL ECX, 1
MOV [EDX], EFX		; guardo la cant de unos en donde estaba el numero
SYS 0X02		; leo la cant de unos

ADD EDX, 4		; si ya termino de leer un numero el puntero apunta al siguiente 
JMP CICLONUMERO 

;---------FIN CICLO EXTERNO

FIN:
STOP
