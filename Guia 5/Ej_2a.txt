SLEN: PUSH BP
MOV BP, SP
PUSH EDX

MOV ECX, 0      ; cont = 0
MOV EDX, [BP + 4] ; EDX = P*

SIGUE_CICLO_SLEN: CMP [EDX], 0  (P*) == 'null'
JZ FIN_CICLO

ADD EDX, 1 ; (P*)++
ADD ECX, 1 ; cont++
JUMP SIGUE_CICLO_SLEN

FIN_CICLO_SLEN: POP EDX
MOV SP, BP
POP BP
RET
