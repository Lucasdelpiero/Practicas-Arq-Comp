; PUSH <PUNTERO_1>
; PUSH <PUNTERO_2>
; ADD SP, 4
; COPIA UNA CADENA DE CARACTERES DEL P_2 AL P_1

SCPY: PUSH BP
MOV BP, SP
PUSH EAX
PUSH EBX

MOV EAX, [BP + 8] ; EAX = P_1
MOV EBX, [BP + 4] ; EBX = P_2

SIGUE_CICLO_SCPY: CMP [EBX], 0 ; P_2 != NULL
JZ FIN_CICLO_SCPY

MOV [EAX], [EBX]; (P_1*) = (P_2*)
ADD EAX, 1	; P_1++
ADD EBX, 1	; P_2++

JMP SIGUE_CICLO_SCPY

FIN_CICLO_SCPY: MOV [EAX], 0 ; Se agrega el 'null' al final de la palabra
POP EBX
POP EAX
MOV SP, BP
POP BP
RET
