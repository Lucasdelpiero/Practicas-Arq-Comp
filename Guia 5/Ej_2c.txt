; PUSH <PUNTERO_1>
; PUSH <PUNTERO_2>
; ADD SP, 4
; DEVUELVE EN EAX UN 0 O UNA DIFERENCIA DEPENDIENDO SI SON IGUALES O DIF

SCMP: PUSH BP
MOV BP, SP
PUSH EBX
PUSH ECX

MOV EBX, [BP + 8] ; EBX = P_1*
MOV ECX, [BP + 4] ; ECX = P_2*

SIGUE_CICLO_SCMP: CMP [EBX], [ECX]  ; (P_1*) == (P_2*)
JNZ SON_DISTINTOS_SCMP

CMP [EBX], 0 	  ; (P_1*) == 'null'
JZ SON_IGUALES_SCMP

ADD EBX, 1	  ;P_1++
ADD ECX, 1	  ;P_2++
JMP SIGUE_CICLO_SCMP

SON_DISTINTOS_SCMP: MOV EAX, [EBX]
SUB EAX, [ECX]	dif queda en EAX
JMP FIN_SCMP

SON_IGUALES_SCMP: MOV EAX, 0

FIN_SCMP: POP ECX
POP EBX
MOV SP, BP
POP BP
RET
