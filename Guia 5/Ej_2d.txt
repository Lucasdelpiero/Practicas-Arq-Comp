; PUSH <PUNTERO_1>
; PUSH <PUNTERO_2>
; ADD SP, 4
; CONCATENA LA PALABRA 1 CON LA PALABRA 2

PUSH BP
MOV BP, SP
PUSH EAX
PUSH EBX
PUSH ECX

MOV EAX, [BP + 8] ; EAX = P_1
MOV EBX, [BP + 4] ; EBX = P_2

PUSH EAX
CALL SLEN	; ECX = LARGO DE PALABRA_1
ADD SP, 4
ADD EAX, ECX ; EAX APUNTA A FINAL DE PALABRA

SIGUE_CICLO_SCAT CMP [EBX], 0
JZ FIN_SCAT

MOV [EAX], [EBX] ; P_1* = P_2*
ADD EAX, 1	 ; P_1++
ADD EBX, 1	 ; P_2++

JMP SIGUE_CICLO_SCAT

FIN_SCAT: MOV [EAX], 0  ; AGREGA 'NULL' AL FINAL DE PALABRA

POP ECX
POP EBX
POP EAX
MOV SP, BP
POP BP
RET
